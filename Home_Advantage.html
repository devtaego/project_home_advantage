import pandas as pd
import time
from datetime import datetime, timedelta

# -----------------------------------
# 2025년 3월 ~ 10월 KBO 경기 수집
# -----------------------------------

all_rows = []

# 2025-03-01 ~ 2025-10-31
start_date = datetime(2025, 3, 1)
end_date = datetime(2025, 10, 31)

current_date = start_date

while current_date <= end_date:
    
    date_str = current_date.strftime("%Y%m%d")
    
    try:
        # 해당 날짜 경기 일정
        games = get_game_schedule.by_date(date_str)
        
        # 해당 날짜 경기 상세정보
        day_data = get_game_data.fetch_date(date_str, delay=0.3)
        
        # 상세 경기 데이터를 ID로 빠르게 찾기
        detail_map = {
            detail["id"]: detail
            for detail in day_data
        }
        
        for game in games:
            
            game_id = game.get("G_ID", "")
            
            # 경기 ID가 이상하면 건너뛰기
            if len(game_id) < 9:
                continue
            
            # 취소 경기 제외
            if str(game.get("CANCEL_SC_ID", "0")) != "0":
                continue
            
            # 20250913HTLG0 → HTLG0
            game_code = game_id[8:]
            
            # 상세 데이터 ID
            detail_id = f"{date_str}_{game_code}"
            
            detail = detail_map.get(detail_id)
            
            if detail is None:
                continue
            
            scoreboard = detail["contents"].get("scoreboard", {})
            
            away_rheb = scoreboard.get("away_RHEB", [])
            home_rheb = scoreboard.get("home_RHEB", [])
            
            # 점수 데이터가 없으면 경기 미진행으로 판단
            if not away_rheb or not home_rheb:
                continue
            
            away_score = int(away_rheb[0])
            home_score = int(home_rheb[0])
            
            # 경기 결과
            if home_score > away_score:
                result = "HOME_WIN"
            elif home_score < away_score:
                result = "AWAY_WIN"
            else:
                result = "DRAW"
            
            all_rows.append({
                "date": date_str,
                "away_team": game.get("AWAY_NM"),
                "home_team": game.get("HOME_NM"),
                "away_score": away_score,
                "home_score": home_score,
                "score_diff": home_score - away_score,
                "result": result
            })
        
        print(f"{date_str} 완료 | 현재 {len(all_rows)}경기")
        
    except Exception as e:
        print(f"{date_str} 오류 → {e}")
    
    current_date += timedelta(days=1)
